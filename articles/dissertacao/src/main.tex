% main.tex -- single entry point for the dissertation (Vitor H. O. Silva, UFV/PPGCC).
%
% THREE builds, ONE source. Two entry files, and this is the main one:
%
%   main.tex       -> the DEFENSE PDF. `make` / `make defense` -> build/main.pdf,
%                     copied to ./dissertacao.pdf. This is the reading copy.
%   main.tex+final -> the AcademicoPG deposit body. `make final` -> build/main_final.pdf.
%   main_ppgc.tex  -> the SAME defense PDF plus the approval sheet. `make ppgc` ->
%                     build/main_ppgc.pdf. That is its ONLY difference from main.tex
%                     (author decision 2026-07-28); it is a four-line file that sets one
%                     switch and reads this one, so the two cannot drift.
%
% Two switches, both defined below, both following the same \ifdefined pattern so a
% command-line \def works without the nested-\if scanning problem:
%   \ifdefensebuild   front matter on (defense) or off (AcademicoPG deposit)
%   \ifapprovalsheet  approval-sheet placeholder on (ppgc) or off (everything else)
%
% ONE source, TWO front-matter modes (UFV_COMPLIANCE §1). The mode is chosen by the
% \ifdefensebuild switch, defined HERE by default as the DEFENSE build:
%
%   DEFENSE build (default): full front matter (cover/folha de rosto, approval
%     placeholder, Resumo/Abstract, lists, sumario) -- the PDF the banca reads.
%     Just compile main.tex normally (this is what `make` / `make defense` does).
%
%   FINAL AcademicoPG build: body only (lists -> sumario -> body); the AcademicoPG
%     system generates cover/ficha/resumo/abstract from web forms. Produced WITHOUT
%     a second main file, by pre-setting the switch to false on the command line:
%       pdflatex "\newif\ifdefensebuild\defensebuildfalse\input{main.tex}"
%     `make final` does exactly this. The body page counter for that build is
%     \finalbuildfirstpage below (tune against the AcademicoPG RASCUNHO PDF).
%
% Build recipe + TeX tree notes: ../src_utils/README_SRC.md. Output goes to build/;
% the current defense PDF is copied to ./dissertacao.pdf at the src root by `make`.
% ============================================================================

% Default mode = DEFENSE. `make final` sets \FINALBUILD on the command line
% (\def\FINALBUILD{}) before reading this file; absent that macro, this is the
% defense build. This avoids a nested-\if scanning problem that \ifdefined has
% when the switch is pre-set on the command line.
\newif\ifdefensebuild
\ifdefined\FINALBUILD \defensebuildfalse \else \defensebuildtrue \fi

% Approval sheet: OFF here, ON in main_ppgc.tex (author decision 2026-07-28). See the
% header comment above. main_ppgc.tex sets \APPROVALSHEET before reading this file, so
% the same \ifdefined guard pattern applies.
\newif\ifapprovalsheet
\ifdefined\APPROVALSHEET \approvalsheettrue \else \approvalsheetfalse \fi

% Page number of the FIRST BODY PAGE in the final build. UFV_COMPLIANCE §4.4: pre-textual
% pages are counted but not numbered, so the first body page prints
% (number of pre-textual pages) + 1.
%
% [round6, C-1, UFV compliance persona 13] CORRECTED 11 -> 8, and the defect is worth
% naming because the comment above it was right the whole time.
% MEASURED, by finding the first page whose header carries a digit in each render and
% reading the printed number:
%   defense (main.pdf)      10 pre-textual pages, first body page is physical p.11, prints 11  OK
%   ppgc    (main_ppgc.pdf) 11 pre-textual pages, physical p.12, prints 12                     OK
%   final   (main_final.pdf) 7 pre-textual pages, physical p.8,  printed 11                    WRONG
% The AcademicoPG deposit build has NO folha de rosto, approval sheet, Resumo or Abstract --
% the system generates those from web forms -- so it has three fewer pre-textual pages than
% the defense build. This counter was hardcoded to the DEFENSE build's offset, and every
% subsequent page inherited the three-page error: the deposit PDF's printed numbers ran
% three ahead of its physical pages throughout, which is a numbering non-compliance on the
% one build that is actually deposited.
% This is only reachable in the final build (0_main.tex:412 sets the counter inside the
% \ifdefensebuild-false branch), which is why the defense and ppgc builds were correct and
% nobody noticed. The [VERIFY] flag it carried asked for tuning against the RASCUNHO PDF and
% was the right instinct; what it did not say is that the value was wrong on its own terms.
\newcommand{\finalbuildfirstpage}{8}
